#include "HybridRMQ.h"
#include "SimpleTest/SimpleTest.h"
#include <cmath>
#include "SparseTableRMQ.h"
using namespace std;

HybridRMQ::HybridRMQ(const RMQEntry* elems, size_t numElems) {
    this->elems = elems;
    this->numElems = numElems;

    if (numElems == 0) return;

    // Compute block size and number of blocks in the array
    block_size = max(1, (int)floor(log2(numElems)));
    size_t num_blocks = (numElems + block_size - 1) / block_size;
    minimums.resize(num_blocks);

    // Compute minimums for each block by linear scan
    for (size_t i = 0; i < num_blocks; i++) {
        size_t block_start = i * block_size;
        size_t block_end = min(block_start + block_size, numElems);
        size_t min_index = block_start;

        for (size_t j = block_start + 1; j < block_end; j++) {
            if (elems[j] < elems[min_index]) {
                min_index = j;
            }
        }

        minimum_indexes[i] = min_index;
        minimums[i] = elems[min_index];
    }

    // Feed minimums into a sparse table
    sparse_table = SparseTableRMQ(minimums.data(), minimums.size());
}


HybridRMQ::~HybridRMQ() {
}

size_t HybridRMQ::rmq(size_t low, size_t high) const {
    // Calculate block indices for the query range
    size_t left_block = low / block_size;
    size_t right_block = high / block_size;

    // Query range is within one block
    if (left_block == right_block) {
        size_t min_index = low;
        for (size_t i = low + 1; i <= high; i++) {
            if (elems[i] < elems[min_index]) {
                min_index = i;
            }
        }
        return min_index;
    }

    // Query range spans multiple blocks
    // Find minimum from the left partial block
    size_t min_index = low;
    size_t left_block_end = min((left_block + 1) * block_size, numElems) - 1;

    for (size_t i = low + 1; i <= left_block_end; i++) {
        if (elems[i] < elems[min_index]) {
            min_index = i;
        }
    }

    // Find minimum from the right partial block
    size_t right_block_start = right_block * block_size;

    for (size_t i = right_block_start; i <= high; i++) {
        if (elems[i] < elems[min_index]) {
            min_index = i;
        }
    }
    
    // Find minimum from the full blocks in between using the sparse table
    if (left_block + 1 <= right_block - 1) {
        size_t sparse_min_index = sparse_table.rmq(left_block + 1, right_block - 1);
    }
}


/* Feel free to add additional test cases here. These are basic checks for correctness
 * that do not account for all possible use cases.
 */
#include "SegmentTreeRMQ.h"
#include <random>
#include <algorithm>

PROVIDED_TEST("Works with a single-element array.") {
    vector<RMQEntry> array = { RMQEntry(137) };
    
    HybridRMQ rmq(array.data(), array.size());
    
    EXPECT_EQUAL(rmq.rmq(0, 0), 0);
}

PROVIDED_TEST("Works with a four-element array.") {
    vector<RMQEntry> array = { RMQEntry(1), RMQEntry(0), RMQEntry(2), RMQEntry(3) };
    
    HybridRMQ rmq(array.data(), array.size());
    
    /* Length one. */
    EXPECT_EQUAL(rmq.rmq(0, 0), 0);
    EXPECT_EQUAL(rmq.rmq(1, 1), 1);
    EXPECT_EQUAL(rmq.rmq(2, 2), 2);
    EXPECT_EQUAL(rmq.rmq(3, 3), 3);
    
    /* Length two. */
    EXPECT_EQUAL(rmq.rmq(0, 1), 1);
    EXPECT_EQUAL(rmq.rmq(1, 2), 1);
    EXPECT_EQUAL(rmq.rmq(2, 3), 2);
    
    /* Length three. */
    EXPECT_EQUAL(rmq.rmq(0, 2), 1);
    EXPECT_EQUAL(rmq.rmq(1, 3), 1);
    
    /* Length four. */
    EXPECT_EQUAL(rmq.rmq(0, 3), 1);
}

PROVIDED_TEST("Works with 1,000-element array.") {
    mt19937 generator(137); // Consistent random values
    
    /* Get a permutation of 0 ... 999. */
    vector<RMQEntry> array(1000);
    for (size_t i = 0; i < array.size(); i++) {
        array[i] = RMQEntry(i);
    }
    shuffle(array.begin(), array.end(), generator);
    
    /* Build an RMQ structure and a reference RMQ structure. */
    HybridRMQ rmq(array.data(), array.size());
    SegmentTreeRMQ ref(array.data(), array.size());
    
    /* Do all possible RMQs. */
    for (size_t i = 0; i < array.size(); i++) {
        for (size_t j = i; j < array.size(); j++) {
            EXPECT_EQUAL(rmq.rmq(i, j), ref.rmq(i, j));
        }
    }
}
